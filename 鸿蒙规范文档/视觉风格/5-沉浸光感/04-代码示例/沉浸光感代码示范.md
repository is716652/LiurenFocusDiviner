# 沉浸光感代码示范

> 本示例基于 HarmonyOS NEXT ArkTS / ArkUI 沉浸式系统材质 API 整理。
> 核心模块：`import { uiMaterial } from '@kit.ArkUI'`。

---

## 一、基础用法：为组件设置系统材质

```typescript
import { uiMaterial } from '@kit.ArkUI';

@Entry
@Component
struct BasicMaterialPage {
  build() {
    Stack() {
      // 复杂背景图
      Image($r('app.media.img'))
        .width('100%')
        .height('100%')
        .objectFit(ImageFit.Cover)

      Column() {
        Text('沉浸光感')
          .fontColor(Color.White)
      }
      .width(328)
      .height(56)
      .borderRadius(28)
      .justifyContent(FlexAlign.Center)
      .systemMaterial(new uiMaterial.ImmersiveMaterial({
        style: uiMaterial.ImmersiveStyle.ULTRA_THIN,
      }))
    }
    .width('100%')
    .height('100%')
  }
}
```

## 二、应用级开关：根据 MaterialState 判断是否可用

```typescript
import { uiMaterial } from '@kit.ArkUI';

@Entry
@Component
struct MaterialStatePage {
  private info: uiMaterial.MaterialInfo = uiMaterial.getMaterialInfo();

  build() {
    Column() {
      Text(`MaterialState: ${this.info.state}`)
        .fontSize(16)
        .margin({ bottom: 10 })
      Text(`MaterialType: ${this.info.type}`)
        .fontSize(16)
        .margin({ bottom: 20 })

      if (this.info.state === uiMaterial.MaterialState.ENABLE) {
        Button('使用沉浸式系统材质')
          .backgroundColor(Color.Transparent)
          .systemMaterial(new uiMaterial.ImmersiveMaterial({
            style: uiMaterial.ImmersiveStyle.ULTRA_THIN
          }))
          .fontColor(Color.Blue)
          .margin({ bottom: 10 })

        Select([{ value: '选项1' }, { value: '选项2' }])
          .value('选择')
          .margin({ bottom: 10 })

        Select([{ value: '选项1' }, { value: '选项2' }])
          .value('选择（已关闭材质）')
          .systemMaterial(uiMaterial.Material.empty)
      }
    }
    .width('100%')
    .height('100%')
    .justifyContent(FlexAlign.Center)
    .backgroundImage($r('app.media.img'))
    .backgroundImageSize(ImageSize.FILL)
  }
}
```

## 四、光感交互反馈：触摸时产生流光跟随

```typescript
import { uiMaterial } from '@kit.ArkUI';

@Entry
@Component
struct LightEffectPage {
  @State itemsKey: number[] = [0, 1, 2];
  @State circleRadius: number = 40;
  @State spaceValue: number = 10;
  @State myMaterial: uiMaterial.Material = new uiMaterial.ImmersiveMaterial({
    style: uiMaterial.ImmersiveStyle.ULTRA_THIN,
    interactive: true,
    lightEffect: { color: undefined },
  });

  build() {
    Column() {
      Row() {
        Text('标题')
          .flexGrow(2)
          .fontColor(Color.White)

        Row({ space: this.spaceValue }) {
          ForEach(this.itemsKey, (item: number, index: number) => {
            Row()
              .width(this.circleRadius * 2)
              .height(this.circleRadius * 2)
              .borderRadius(this.circleRadius)
              .systemMaterial(this.myMaterial)
          }, (item: number) => item.toString())
        }
      }
      .justifyContent(FlexAlign.End)
      .backgroundColor(Color.Black)
      .width('100%')
      .padding(20)
    }
    .height('100%')
    .width('100%')
  }
}
```

struct SheetMaterialPage {

            height: this.sheetHeight,
            backgroundColor: Color.Transparent,
            systemMaterial: this.myMaterial
          })
      }
      .justifyContent(FlexAlign.Center)
      .width('100%')
      .height('100%')
    }
  }

在 `ENABLE` 模式下，部分组件会默认开启沉浸式系统材质。如需单独关闭某个组件的系统材质，可设置：

```typescript
.someComponentProperty()
.systemMaterial(uiMaterial.Material.empty)
```

## 九、窗口沉浸式配置（可选）

如果需要内容延伸至状态栏、导航栏，可在 `EntryAbility` 中配置窗口属性：

```typescript
// entry/src/main/ets/entryability/EntryAbility.ets
import { UIAbility } from '@kit.AbilityKit';
import { window } from '@kit.ArkUI';

export default class EntryAbility extends UIAbility {
  onWindowStageCreate(windowStage: window.WindowStage): void {
    windowStage.loadContent('pages/Index', (err) => {
      if (err && err.code) {
        console.error('Failed to load content:', JSON.stringify(err));
        return;
      }

      const mainWindow = windowStage.getMainWindowSync();
      mainWindow.setWindowLayoutFullScreen(true);
      mainWindow.setWindowBackgroundColor('#00000000');
      mainWindow.setWindowSystemBarProperties({
        statusBarColor: '#00000000',
        navigationBarColor: '#00000000',
        statusBarContentColor: '#FFFFFF',
        navigationBarContentColor: '#FFFFFF'
      });
    });
  }
}
```

> 注意：窗口沉浸式配置与 `systemMaterial` 是两套能力。前者负责布局延伸至系统栏，后者负责组件材质效果。根据实际需求组合使用。

## 十、常见误区与建议

| 误区 | 建议 |
|------|------|
| 所有组件都加材质 | 只在需要层级分离或背景重叠的场景使用 |
| 用材质替代对比度 | 仍需保证文字/图标可读性，必要时开启 `colorInvert` |
| 为三档强度分别写代码 | 只需定义一次材质，系统会自动映射 |
| 忽略弱档效果 | 设计稿应以均衡档为基准，但需在弱档下验证可读性 |
| 过度使用 `lightEffect` | 交互光效应自然，避免喧宾夺主 |
